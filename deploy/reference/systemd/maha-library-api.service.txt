# /etc/systemd/system/maha-library-api.service
[Unit]
Description=MAHA local book uploads and catalog
After=network.target
RequiresMountsFor=/mnt/data/survival-library/imported-books
[Service]
User=vhenderson
Group=vhenderson
ExecStart=/usr/bin/python3 /opt/maha-library-api/server.py
Restart=on-failure
Environment=QT_QPA_PLATFORM=offscreen
Environment=MAHA_IMPORT_ROOT=/mnt/data/survival-library/imported-books
UMask=0022
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
IPAddressDeny=any
IPAddressAllow=localhost
ReadWritePaths=/mnt/data/survival-library/imported-books
MemoryMax=2G
TasksMax=128
[Install]
WantedBy=multi-user.target

# /etc/systemd/system/maha-library-api.service.d/accounts.conf
[Service]
Environment=MAHA_REQUIRE_AUTH=1
Environment=MAHA_AUTH_DB=/srv/maha-library-api/accounts.sqlite
Environment=MAHA_PUBLIC_URL=https://192.168.1.64/
ReadWritePaths=/srv/maha-library-api

# /etc/systemd/system/maha-library-api.service.d/music.conf
[Service]
Environment=PYTHONPATH=/opt/maha-library-api/music-vendor
